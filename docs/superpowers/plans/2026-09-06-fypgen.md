# fypgen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `fypgen` — a free-forever, local, one-command FYP report generator for Nigerian students (Vite React SPA + Hono local server + `npx` CLI, BYO AI key, DOCX/PPTX export).

**Architecture:** A single npm package. `bin/fypgen.js` boots a Hono server on a free localhost port and opens the browser; the server serves the built Vite SPA and proxies streaming AI calls (so API keys never hit browser-CORS problems). The SPA holds all product logic: onboarding → project form → outline → chapter-by-chapter streaming generation → review → DOCX/PPTX export. State persists in localStorage.

**Tech Stack:** TypeScript, React + Vite, Hono, zustand (with persist), `docx`, `pptxgenjs`, vitest, Playwright. Providers: Gemini (default), OpenAI, Anthropic.

**Spec:** `docs/superpowers/specs/2026-09-06-fypgen-design.md`

## Global Constraints

- No accounts, no payments, no telemetry, no server-side persistence — the API key is sent per-request, never written to disk, never logged.
- Distribution target: `npx fypgen` (Node ≥ 18, no build step required by the end user).
- Default models: `gemini-2.5-flash` (Gemini), `gpt-4o-mini` (OpenAI), `claude-haiku-4-5-20251001` (Anthropic); overridable in settings.
- Standard Nigerian FYP format: 5 chapters, Times New Roman 12, double-spaced, justified, 1" margins, numbered headings (CHAPTER ONE…), APA 7 references labeled *"AI-generated references — verify before submission."*
- All generation must stream; a dropped stream preserves partial text and offers **Continue**.
- Test commands run from the repo root: `npx vitest run <path>`.
- TDD: every task writes the failing test first, then the minimal implementation, then commits.

```ts
// server/static.ts
import { serveStatic } from '@hono/node-server/serve-static'
import { readFileSync, existsSync } from 'node:fs'
import { join } from 'node:path'
import type { Context } from 'hono'

/** Serves webRoot assets; falls back to index.html so client-side routes resolve. */
export function spaStatic(webRoot: string) {
  const asset = serveStatic({ root: webRoot })
  return async (c: Context, next: () => Promise<void>) => {
    await asset(c, next)
    if (c.res.status === 404) {
      const index = join(webRoot, 'index.html')
      if (existsSync(index)) {
        return c.html(readFileSync(index, 'utf-8'))
      }
    }
  }
}
```

```ts
// server/port.ts
import { createServer } from 'node:net'

export function getFreePort(): Promise<number> {
  return new Promise((resolve, reject) => {
    const srv = createServer()
    srv.unref()
    srv.once('error', reject)
    srv.listen(0, () => {
      const addr = srv.address()
      if (addr && typeof addr === 'object') resolve(addr.port)
      else reject(new Error('no port'))
      srv.close()
    })
  })
}
```

```ts
// server/openBrowser.ts
import { spawn } from 'node:child_process'
import process from 'node:process'

export function commandFor(platform: NodeJS.Platform): string[] {
  if (platform === 'darwin') return ['open']
  if (platform === 'win32') return ['cmd', '/c', 'start', '']
  return ['xdg-open']
}

export async function openBrowser(url: string): Promise<void> {
  const [cmd, ...args] = commandFor(process.platform)
  const child = spawn(cmd, [...args, url], { stdio: 'ignore', detached: true })
  child.unref()
}
```

```ts
// server/boot.ts
import { serve } from '@hono/node-server'
import { createApp } from './index.js'
import { getFreePort } from './port.js'
import type { Hono } from 'hono'

export interface BootOptions {
  webRoot: string
  port?: number
  routes?: (app: Hono) => void
}

export async function bootServer(opts: BootOptions) {
  const port = opts.port ?? (await getFreePort())
  const app = createApp({ webRoot: opts.webRoot })
  opts.routes?.(app)
  const server = serve({ fetch: app.fetch, port, hostname: '127.0.0.1' })
  return {
    port,
    close: () => new Promise<void>((resolve) => server.close(() => resolve())),
  }
}
```

```ts
// server/providers/sse.ts
/** Parses an SSE Response body; `extract` maps each data payload to text (or undefined to skip). */
export async function* parseSSE(
  response: Response,
  extract: (data: string) => string | undefined
): AsyncGenerator<string> {
  if (!response.body) return
  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    let idx: number
    while ((idx = buffer.indexOf('\n\n')) !== -1) {
      const event = buffer.slice(0, idx)
      buffer = buffer.slice(idx + 2)
      for (const line of event.split('\n')) {
        if (line.startsWith('data: ')) {
          const text = extract(line.slice(6))
          if (text !== undefined) yield text
        }
      }
    }
  }
}
```

```ts
// server/providers/gemini.ts
import type { AIProvider } from './types.js'
import { parseSSE } from './sse.js'
import { DEFAULT_MODELS } from '../../shared/constants.js'

const BASE = 'https://generativelanguage.googleapis.com/v1beta/models'

export const gemini: AIProvider = {
  id: 'gemini',
  defaultModel: DEFAULT_MODELS.gemini,

  async testKey(apiKey) {
    const res = await fetch(`${BASE}/${this.defaultModel}:generateContent`, {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'x-goog-api-key': apiKey },
      body: JSON.stringify({ contents: [{ role: 'user', parts: [{ text: 'hi' }] }] }),
    })
    if (res.ok) return { ok: true }
    const body = (await res.json().catch(() => ({}))) as { error?: { message?: string } }
    return { ok: false, error: body.error?.message ?? `HTTP ${res.status}` }
  },

  async *stream({ apiKey, model, system, prompt }) {
    const m = model ?? this.defaultModel
    const res = await fetch(`${BASE}/${m}:streamGenerateContent?alt=sse`, {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'x-goog-api-key': apiKey },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: system }] },
        contents: [{ role: 'user', parts: [{ text: prompt }] }],
      }),
    })
    if (!res.ok) {
      const body = (await res.json().catch(() => ({}))) as { error?: { message?: string } }
      throw new Error(body.error?.message ?? `Gemini error HTTP ${res.status}`)
    }
    yield* parseSSE(res, (data) => {
      const j = JSON.parse(data) as { candidates?: { content?: { parts?: { text?: string }[] } }[] }
      return j.candidates?.[0]?.content?.parts?.[0]?.text
    })
  },
}
```

```ts
export const openai: AIProvider = {

    const res = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',

    if (res.ok) return { ok: true }
    const body = (await res.json().catch(() => ({}))) as { error?: { message?: string } }

      body: JSON.stringify({

        stream: true,
        messages: [
          { role: 'system', content: system },
          { role: 'user', content: prompt },
        ],
      }),

      throw new Error(body.error?.message ?? `OpenAI error HTTP ${res.status}`)
    }
    yield* parseSSE(res, (data) => {
      if (data === '[DONE]') return undefined
```

```ts
// server/routes.ts
import type { Hono } from 'hono'
import { stream } from 'hono/streaming'
import { REGISTRY } from './providers/registry.js'
import { PROVIDERS } from '../shared/constants.js'

interface GenerateBody {
  provider?: string
  apiKey?: string
  model?: string
  system?: string
  prompt?: string
}

export function registerAIRoutes(app: Hono) {
  app.post('/api/test-key', async (c) => {
    const body = (await c.req.json().catch(() => ({}))) as GenerateBody
    if (!body.provider || !PROVIDERS.includes(body.provider as never)) {
      return c.json({ ok: false, error: 'Unknown provider' }, 400)
    }
    if (!body.apiKey) return c.json({ ok: false, error: 'Missing API key' }, 400)
    const provider = REGISTRY[body.provider]
    const result = await provider.testKey(body.apiKey)
    return c.json(result, result.ok ? 200 : 401)
  })

  app.post('/api/generate', async (c) => {
    const body = (await c.req.json().catch(() => ({}))) as GenerateBody
    if (!body.provider || !PROVIDERS.includes(body.provider as never)) {
      return c.json({ error: 'Unknown provider' }, 400)
    }
    if (!body.apiKey) return c.json({ error: 'Missing API key' }, 400)
    if (!body.system || !body.prompt) return c.json({ error: 'Missing system or prompt' }, 400)
    const provider = REGISTRY[body.provider]
    try {
      return stream(c, async (s) => {
        for await (const chunk of provider.stream({
          apiKey: body.apiKey!,
          model: body.model,
          system: body.system!,
          prompt: body.prompt!,
        })) {
          await s.write(chunk)
        }
      })
    } catch (err) {
      return c.json({ error: err instanceof Error ? err.message : 'Generation failed' }, 502)
    }
  })
}
```

```ts
// web/src/state/store.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { ProviderId } from '@shared/constants'

export interface Settings { provider: ProviderId; apiKey: string; model?: string }
export interface ProjectInfo {
  topic: string; studentName: string; matric: string; university: string
  department: string; supervisor: string; degree: string; session: string
}
export interface Section { heading: string; description: string }
export type ChapterStatus = 'pending' | 'streaming' | 'done'
export interface Chapter {
  number: number; title: string; sections: Section[]
  status: ChapterStatus; text: string
}

export const EMPTY_PROJECT: ProjectInfo = {
  topic: '', studentName: '', matric: '', university: '',
  department: '', supervisor: '', degree: '', session: '',
}

export const STANDARD_CHAPTERS = [
  'INTRODUCTION', 'LITERATURE REVIEW', 'METHODOLOGY',
  'RESULTS AND ANALYSIS', 'CONCLUSION AND RECOMMENDATIONS',
]

export function initialChapterState(): Chapter[] {
  return STANDARD_CHAPTERS.map((title, i) => ({
    number: i + 1, title, sections: [], status: 'pending' as ChapterStatus, text: '',
  }))
}

interface AppState {
  settings: Settings
  project: ProjectInfo
  chapters: Chapter[]
  abstract: string
  setSettings: (s: Settings) => void
  setProject: (p: Partial<ProjectInfo>) => void
  setOutline: (chapters: Chapter[], abstract: string) => void
  updateChapter: (number: number, patch: Partial<Chapter>) => void
  appendChapterText: (number: number, chunk: string) => void
  reset: () => void
}

export const useAppStore = create<AppState>()(
  persist(
    (set) => ({
      settings: { provider: 'gemini', apiKey: '' },
      project: EMPTY_PROJECT,
      chapters: initialChapterState(),
      abstract: '',
      setSettings: (s) => set({ settings: s }),
      setProject: (p) => set((st) => ({ project: { ...st.project, ...p } })),
      setOutline: (chapters, abstract) => set({ chapters, abstract }),
      updateChapter: (number, patch) =>
        set((st) => ({
          chapters: st.chapters.map((c) => (c.number === number ? { ...c, ...patch } : c)),
        })),
      appendChapterText: (number, chunk) =>
        set((st) => ({
          chapters: st.chapters.map((c) => (c.number === number ? { ...c, text: c.text + chunk } : c)),
        })),
      reset: () =>
        set({ project: EMPTY_PROJECT, chapters: initialChapterState(), abstract: '' }),
    }),
    { name: 'fypgen-state' }
  )
)

export const useSettings = () => useAppStore((s) => s.settings)
export const useProject = () => useAppStore((s) => s.project)
export const useChapters = () => useAppStore((s) => s.chapters)
export const useAbstract = () => useAppStore((s) => s.abstract)
```

```ts
import type { Settings } from './state/store'

export class ApiError extends Error {
  status: number
  constructor(status: number, message: string) {
    super(message)
    this.status = status
  }
}

export async function testKey(settings: Settings): Promise<{ ok: boolean; error?: string }> {
  const res = await fetch('/api/test-key', {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ provider: settings.provider, apiKey: settings.apiKey, model: settings.model }),
  })
  return res.json()
}

export async function generateStream(
  settings: Settings,
  system: string,
  prompt: string,
  onChunk: (chunk: string) => void
): Promise<string> {
  const res = await fetch('/api/generate', {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({
      provider: settings.provider, apiKey: settings.apiKey, model: settings.model,
      system, prompt,
    }),
  })
  if (!res.ok || !res.body) {
    const body = (await res.json().catch(() => ({ error: `HTTP ${res.status}` }))) as { error?: string }
    throw new ApiError(res.status, body.error ?? `HTTP ${res.status}`)
  }
  const reader = res.body.getReader()
  const decoder = new TextDecoder()
  let full = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    const chunk = decoder.decode(value, { stream: true })
    full += chunk
    onChunk(chunk)
  }
  return full
}
```

```ts
// web/src/ai/parseOutline.ts
import type { Chapter, ChapterStatus } from '../state/store'

interface RawOutline {
  abstract?: string
  chapters?: { title?: string; sections?: { heading?: string; description?: string }[] }[]
}

/** Parses the model's outline JSON, tolerating fenced code blocks and surrounding prose. */
export function parseOutline(raw: string): { chapters: Chapter[]; abstract: string } {
  const candidates: string[] = []
  const fenced = raw.match(/```(?:json)?\s*([\s\S]*?)```/)
  if (fenced) candidates.push(fenced[1])
  const first = raw.indexOf('{')
  const last = raw.lastIndexOf('}')
  if (first !== -1 && last > first) candidates.push(raw.slice(first, last + 1))
  candidates.push(raw)

  for (const candidate of candidates) {
    try {
      const parsed = JSON.parse(candidate.trim()) as RawOutline
      if (!parsed.chapters?.length) continue
      const chapters: Chapter[] = parsed.chapters.slice(0, 5).map((c, i) => ({
        number: i + 1,
        title: (c.title ?? `Chapter ${i + 1}`).toUpperCase(),
        sections: (c.sections ?? [])
          .filter((s) => s.heading)
          .map((s) => ({ heading: s.heading!, description: s.description ?? '' })),
        status: 'pending' as ChapterStatus,
        text: '',
      }))
      return { chapters, abstract: parsed.abstract ?? '' }
    } catch {
      // try next candidate
    }
  }
  throw new Error('Model did not return valid JSON')
}
```

```ts
// web/src/useChapterGeneration.ts
import { useCallback, useState } from 'react'
import { useAppStore } from './state/store'
import { generateStream } from './api'
import { systemPrompt, chapterPrompt } from './prompts'

export function useChapterGeneration() {
  const [busy, setBusy] = useState(false)
  const [continuing, setContinuing] = useState<number | null>(null)

  const generate = useCallback(
    async (number: number, nudge?: string, continuation?: string) => {
      const { settings, project, chapters, updateChapter, appendChapterText } = useAppStore.getState()
      const chapter = chapters.find((c) => c.number === number)
      if (!chapter) return
      updateChapter(number, { status: 'streaming', ...(continuation ? {} : { text: '' }) })
      setBusy(true)
      try {
        await generateStream(
          settings,
          systemPrompt(),
          chapterPrompt(project, chapters, chapter, nudge, continuation),
          (chunk) => useAppStore.getState().appendChapterText(number, chunk)
        )
        updateChapter(number, { status: 'done' })
        setContinuing(null)
      } catch (err) {
        // partial text is already in the store; keep status 'streaming' so Continue shows
        if (!useAppStore.getState().chapters.find((c) => c.number === number)?.text) {
          updateChapter(number, { status: 'pending' })
        } else {
          setContinuing(number)
        }
        throw err
      } finally {
        setBusy(false)
      }
    },
    []
  )

  const continueChapter = useCallback(
    async (number: number) => {
      const chapter = useAppStore.getState().chapters.find((c) => c.number === number)
      if (!chapter?.text) return
      await generate(number, undefined, chapter.text)
    },
    [generate]
  )

  const generateAll = useCallback(async () => {
    setBusy(true)
    try {
      const pending = useAppStore.getState().chapters.filter((c) => c.status !== 'done')
      for (const ch of pending) {
        await generate(ch.number)
      }
    } finally {
      setBusy(false)
    }
  }, [generate])

  return { generate, generateAll, busy, continuing, continueChapter }
}
```

```ts
// web/src/export/docx.ts
import {
  AlignmentType, Document, HeadingLevel, Packer, Paragraph, TextRun,
} from 'docx'
import type { Chapter, ProjectInfo } from '../state/store'

export const REFERENCES_LABEL = 'AI-generated references — verify before submission.'

/** Extracts any "References" section from chapter texts; synthesizes one if absent. */
export function referencesSection(chapters: Chapter[]): string {
  const collected: string[] = []
  for (const ch of chapters) {
    const idx = ch.text.lastIndexOf('References')
    if (idx !== -1 && idx < ch.text.length) {
      collected.push(ch.text.slice(idx + 'References'.length).trim())
    }
  }
  const body = collected.length
    ? collected.join('\n\n')
    : 'No references were generated. Add your references here in APA 7 format.'
  return `${body}\n\n${REFERENCES_LABEL}`
}

const body = (text: string, opts: { bold?: boolean; heading?: (typeof HeadingLevel)[keyof typeof HeadingLevel]; align?: (typeof AlignmentType)[keyof typeof AlignmentType] } = {}) =>
  new Paragraph({
    heading: opts.heading,
    alignment: opts.align ?? AlignmentType.JUSTIFIED,
    spacing: { line: 480 }, // double spacing
    children: [new TextRun({ text, bold: opts.bold, font: 'Times New Roman', size: 24 })], // 12pt = 24 half-points
  })

export async function buildDocx(
  project: ProjectInfo,
  chapters: Chapter[],
  abstract: string
): Promise<Blob> {
  const children: Paragraph[] = []

  // Title page
  children.push(body(project.university.toUpperCase(), { bold: true, align: AlignmentType.CENTER, heading: HeadingLevel.HEADING_1 }))
  children.push(body(project.department, { align: AlignmentType.CENTER }))
  children.push(body('', {}))
  children.push(body(project.topic, { bold: true, align: AlignmentType.CENTER }))
  children.push(body('by', { align: AlignmentType.CENTER }))
  children.push(body(`${project.studentName} (${project.matric})`, { align: AlignmentType.CENTER }))
  children.push(body('', {}))
  children.push(body(`A ${project.degree} Final Year Project Report`))
  children.push(body(`Supervisor: ${project.supervisor}`, { align: AlignmentType.CENTER }))
  children.push(body(project.session, { align: AlignmentType.CENTER }))

  // Abstract + TOC heading
  children.push(body('ABSTRACT', { bold: true, align: AlignmentType.CENTER, heading: HeadingLevel.HEADING_1 }))
  children.push(body(abstract || 'Abstract pending generation.'))
  children.push(body('TABLE OF CONTENTS', { bold: true, align: AlignmentType.CENTER, heading: HeadingLevel.HEADING_1 }))
  for (const ch of chapters) {
    children.push(body(`CHAPTER ${toWord(ch.number)}: ${ch.title}`, { bold: true }))
  }
  children.push(body('REFERENCES', { bold: true }))

  // Chapters
  for (const ch of chapters) {
    if (!ch.text) continue
    for (const line of ch.text.split('\n')) {
      const t = line.trim()
      if (!t) continue
      if (/^CHAPTER\s+/i.test(t)) children.push(body(t.toUpperCase(), { bold: true, heading: HeadingLevel.HEADING_1, align: AlignmentType.LEFT }))
      else if (/^\d+\.\d+/.test(t)) children.push(body(t, { heading: HeadingLevel.HEADING_2, align: AlignmentType.LEFT }))
      else children.push(body(t))
    }
  }

  // References
  children.push(body('REFERENCES', { bold: true, align: AlignmentType.CENTER, heading: HeadingLevel.HEADING_1 }))
  for (const ref of referencesSection(chapters).split('\n')) {
    const t = ref.trim()
    if (t) children.push(body(t, { align: AlignmentType.LEFT }))
  }

  const doc = new Document({
    styles: { default: { document: { run: { font: 'Times New Roman', size: 24 } } } },
    sections: [{ properties: { page: { margin: { top: 1440, bottom: 1440, left: 1440, right: 1440 } } }, children }],
  })
  return Packer.toBlob(doc)
}

const WORDS = ['ONE', 'TWO', 'THREE', 'FOUR', 'FIVE', 'SIX', 'SEVEN', 'EIGHT', 'NINE', 'TEN']
function toWord(n: number): string {
  return WORDS[n - 1] ?? String(n)
}
```

```ts
// web/src/export/pptx.ts
import pptxgen from 'pptxgenjs'
import type { Chapter, ProjectInfo } from '../state/store'

interface SimpleSlide { title: string; body: string[] }

/** Parses defenseQa text into one slide per chapter block. */
export function parseQaSlides(defenseQa: string): SimpleSlide[] {
  const slides: SimpleSlide[] = []
  const blocks = defenseQa.split(/\n(?=Chapter \d)/i)
  for (const block of blocks) {
    const lines = block.split('\n').map((l) => l.trim()).filter(Boolean)
    if (!lines.length) continue
    const questions = lines.filter((l) => /^(Q:|A:)/.test(l))
    if (questions.length) slides.push({ title: `Q&A Prep — ${lines[0]}`, body: questions })
  }
  return slides
}

function slideText(chapters: Chapter[], chapterNumber: number, sectionPrefixes: string[]): string[] {
  const ch = chapters.find((c) => c.number === chapterNumber)
  if (!ch) return []
  return ch.sections
    .filter((s) => sectionPrefixes.some((p) => s.heading.toLowerCase().startsWith(p)))
    .map((s) => s.heading)
}

export async function buildPptx(
  project: ProjectInfo,
  chapters: Chapter[],
  defenseQa: string
): Promise<Blob> {
  const pptx = new pptxgen()
  pptx.layout = 'LAYOUT_16x9'

  const add = (title: string, body: string[]) => {
    const s = pptx.addSlide()
    s.addText(title, { x: 0.5, y: 0.4, fontSize: 28, bold: true, color: '1a1a2e' })
    s.addText(body.map((t) => ({ text: t, options: { bullet: true } })), {
      x: 0.5, y: 1.4, fontSize: 16, color: '333333',
    })
  }

  // 1. Title slide
  const title = pptx.addSlide()
  title.addText(project.topic, { x: 0.5, y: 1.6, fontSize: 30, bold: true, align: 'center', color: '1a1a2e' })
  title.addText(
    [project.studentName, `${project.department}, ${project.university}`, `Supervisor: ${project.supervisor} · ${project.session}`].join('\n'),
    { x: 0.5, y: 3.2, fontSize: 16, align: 'center', color: '555555' }
  )

  // 2–8. Core slides (each falls back to chapter title bullets when sections don't match)
  const ch = (n: number) => chapters.find((c) => c.number === n)
  add('Background & Problem', slideText(chapters, 1, ['background', 'statement of the problem', 'problem']))
  add('Objectives of the Study', slideText(chapters, 1, ['objective', 'aim', 'purpose']))
  add('Scope of the Study', slideText(chapters, 1, ['scope', 'limitation', 'significance']))
  add('Methodology', ch(3)?.sections.map((s) => s.heading) ?? ['System design', 'Implementation', 'Testing'])
  add('Results & Analysis', ch(4)?.sections.map((s) => s.heading) ?? ['Key findings', 'Performance evaluation'])
  add('Conclusion', ch(5)?.sections.filter((s) => /conclusion|summary/.test(s.heading)).map((s) => s.heading) ?? ['Summary of findings'])
  add('Recommendations', ch(5)?.sections.filter((s) => /recommend/.test(s.heading)).map((s) => s.heading) ?? ['Future work'])

  // 9. Thank you
  add('Thank You', ['Questions, corrections and contributions are welcome.'])

  // 10+. Q&A prep appendix
  for (const qa of parseQaSlides(defenseQa)) add(qa.title, qa.body)

  const data = (await pptx.write({ outputType: 'blob' })) as Blob
  return data
}
```

```tsx
// web/src/screens/Export.tsx
import { useState } from 'react'
import { useAppStore, useProject, useChapters, useSettings } from '../state/store'
import { generateStream } from '../api'
import { systemPrompt, defenseQuestionsPrompt } from '../prompts'
import { buildDocx } from '../export/docx'
import { buildPptx } from '../export/pptx'
import { downloadBlob } from '../export/download'

function slug(topic: string): string {
  return topic.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').slice(0, 60) || 'fyp-project'
}

export function Export() {
  const project = useProject()
  const chapters = useChapters()
  const settings = useSettings()
  const abstract = useAppStore((s) => s.abstract)
  const [defenseQa, setDefenseQa] = useState('')
  const [busy, setBusy] = useState('')
  const [error, setError] = useState('')

  async function docx() {
    setBusy('docx')
    setError('')
    try {
      downloadBlob(await buildDocx(project, chapters, abstract), `${slug(project.topic)}.docx`)
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Export failed')
    } finally {
      setBusy('')
    }
  }

  async function pptx() {
    setBusy('pptx')
    setError('')
    try {
      if (!defenseQa) {
        const qa = await generateStream(
          settings, systemPrompt(), defenseQuestionsPrompt(project, chapters), () => {}
        )
        setDefenseQa(qa)
        downloadBlob(await buildPptx(project, chapters, qa), `${slug(project.topic)}-defense.pptx`)
      } else {
        downloadBlob(await buildPptx(project, chapters, defenseQa), `${slug(project.topic)}-defense.pptx`)
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Export failed')
    } finally {
      setBusy('')
    }
  }

  const done = chapters.filter((c) => c.text).length
  return (
    <main className="container">
      <h1>Export</h1>
      <p className="note">{done} of {chapters.length} chapters have content.</p>
      <button disabled={done === 0 || busy !== ''} onClick={() => void docx()}>
        {busy === 'docx' ? 'Building…' : 'Download DOCX'}
      </button>{' '}
      <button disabled={done === 0 || busy !== ''} onClick={() => void pptx()}>
        {busy === 'pptx' ? 'Generating slides…' : 'Download Defense Slides (PPTX)'}
      </button>
      {error && <p role="alert" className="error">{error}</p>}
      {defenseQa && (
        <details style={{ marginTop: '1.5rem' }}>
          <summary>Defense Q&A prep</summary>
          <pre style={{ whiteSpace: 'pre-wrap' }}>{defenseQa}</pre>
        </details>
      )}
      <p className="note">Start a new project: <button className="link" onClick={() => useAppStore.getState().reset()}>reset</button></p>
    </main>
  )
}
```
